    {
        $scope = session('operational_scope');
        $db = db_connect();

        $plans = $db->table('discharge_plans dp')
            ->select('dp.id, dp.folio, dp.voyage_id, rsa.code AS rsa_code, v.annual_number, ve.code AS vessel_code, ve.name AS vessel_name')
            ->join('voyages v', 'v.id = dp.voyage_id')
            ->join('vessels ve', 've.id = v.vessel_id')
            ->join('rsa_records rsa', 'rsa.id = dp.rsa_id', 'left')
            ->where(['v.company_id' => $scope['company_id'], 'v.site_id' => $scope['site_id'], 'dp.status' => 'in_progress'])
            ->orderBy('dp.planned_start_at')
            ->get()->getResultArray();
        $planIds = array_map('intval', array_column($plans, 'id'));
        $voyageIds = array_map('intval', array_column($plans, 'voyage_id'));

        $lines = $db->table('discharge_plan_tanks dpt')
            ->select('dpt.id, dpt.discharge_plan_id, dpt.voyage_tank_id, dpt.species_id, dpt.size_grade_id, dpt.planned_weight_kg, dpt.msc_code, dpt.dolphin_safe_code, dpt.freezing_temperature_c, vt.tank_number, s.code AS species_code, s.name AS species_name, sg.name AS size_name')
            ->join('voyage_tanks vt', 'vt.id = dpt.voyage_tank_id')
            ->join('species s', 's.id = dpt.species_id')
            ->join('size_grades sg', 'sg.id = dpt.size_grade_id', 'left')
            ->whereIn('dpt.discharge_plan_id', $planIds ?: [0])
            ->orderBy('dpt.sequence_number')
            ->get()->getResultArray();

        $tanks = $db->table('voyage_tanks')
            ->select('id, voyage_id, tank_number')
            ->whereIn('voyage_id', $voyageIds ?: [0])
            ->orderBy('tank_number')
            ->get()->getResultArray();

        $tubs = $db->table('tubs')
            ->select('id, code, barcode, status')
            ->where(['company_id' => $scope['company_id'], 'active' => 1, 'status' => 'available'])
            ->orderBy('code')
            ->get()->getResultArray();

        $fillings = $db->table('tub_fillings tf')
            ->select("tf.id, tf.status, tf.started_at, tf.completed_at, dp.folio AS plan_folio, t.code AS tub_code, vt.tank_number, s.name AS species_name, sg.name AS size_name, GROUP_CONCAT(srcvt.tank_number ORDER BY fst.is_primary DESC, srcvt.tank_number SEPARATOR ', ') AS source_tanks", false)
            ->join('discharge_plans dp', 'dp.id = tf.discharge_plan_id')
            ->join('voyages v', 'v.id = dp.voyage_id')
            ->join('tubs t', 't.id = tf.tub_id')

    public function detail(int $planId): array
    {

                'created_by' => $actorUserId,
                'created_at' => $now,
                'updated_at' => $now,

    {

        if (! in_array($source['voyage_status'], ['planned', 'unloading'], true)) {
            throw new DomainException(

            );
        }

        $rsaId = $source['rsa_id'];

        if ($rsaId !== null) {

                    'voyage_id' => (int) $source['voyage_id'],
                    'status' => 'active',
                ])
                ->get()
                ->getRowArray();

            $referenceDate = $this->referenceDate($source['planned_start_at']);

            if (
                $rsa === null
                || ($rsa['issued_on'] !== null && $rsa['issued_on'] > $referenceDate)
                || ($rsa['expires_on'] !== null && $rsa['expires_on'] < $referenceDate)
            ) {

                    'sequence_number' => (int) $line['sequence_number'],

    public function transition(
        int $planId,
        string $target,
        int $actorUserId,
        ?string $reason = null
    ): array {
        $reason = trim((string) $reason);

        if (
            in_array($target, ['canceled', 'draft'], true)
            && (mb_strlen($reason) < 5 || mb_strlen($reason) > 500)
        ) {
            throw new DomainException(
                $target === 'draft'
                    ? 'Regresar un plan a borrador requiere un motivo de 5 a 500 caracteres.'
                    : 'Cancelar un plan requiere un motivo de 5 a 500 caracteres.',
                $target === 'draft'
                    ? 'PLAN_RETURN_DRAFT_REASON_REQUIRED'
                    : 'PLAN_CANCEL_REASON_REQUIRED'
            );
        }

        $this->db->transException(true)->transBegin();

        try {
            $plan = $this->db->query(
                <<<'SQL'
SELECT dp.*, v.company_id, v.site_id, v.status AS voyage_status
FROM discharge_plans dp
INNER JOIN voyages v ON v.id = dp.voyage_id
WHERE dp.id = ?
FOR UPDATE
SQL,
                [$planId]
            )->getRowArray();

            if ($plan === null) {
                throw new DomainException('Plan no encontrado.', 'PLAN_NOT_FOUND');
            }

            WorkflowTransitions::plan()->assertCanTransition(
                (string) $plan['status'],
                $target
            );

            $fillings = $this->db->query(
                'SELECT id, status FROM tub_fillings WHERE discharge_plan_id = ? FOR UPDATE',
                [$planId]
            )->getResultArray();

            if ($target === 'approved') {
                $review = $this->buildReview($plan);

                if (! $review['can_approve']) {
                    $failure = $this->firstFailedValidation($review);
                    throw new DomainException(
                        $failure['message']
                            ?? 'El plan no cumple todas las validaciones para aprobarse.',
                        'PLAN_NOT_READY_FOR_APPROVAL',
                        ['review' => $review]
                    );
                }
            }

            if ($target === 'draft' && $fillings !== []) {
                throw new DomainException(
                    'El plan ya tiene operaciÃ³n y no puede regresar a borrador.',
                    'PLAN_RETURN_DRAFT_HAS_OPERATIONS'
                );
            }

            if ($target === 'finished') {
                foreach ($fillings as $filling) {
                    if (in_array($filling['status'], ['filling', 'pending_weighing'], true)) {
                        throw new DomainException(
                            'No se puede finalizar un plan con tinas en llenado o pendientes de pesaje.',
                            'PLAN_HAS_PENDING_FILLINGS'
                        );
                    }
                }
            }

            if ($target === 'canceled' && $fillings !== []) {
                throw new DomainException(
                    'Un plan que ya tiene operaciÃ³n de tinas no puede cancelarse.',
                    'PLAN_HAS_OPERATIONS'
                );
            }

            $now = date('Y-m-d H:i:s');
            $update = [
                'status' => $target,
                'lock_version' => (int) $plan['lock_version'] + 1,
                'updated_by' => $actorUserId,
                'updated_at' => $now,
            ];

            if ($target === 'approved') {
                $update['approved_by'] = $actorUserId;
                $update['approved_at'] = $now;
            }

            if ($target === 'draft') {
                $update['approved_by'] = null;
                $update['approved_at'] = null;
            }

            $this->db->table('discharge_plans')
                ->where('id', $planId)
                ->update($update);

            if ($target === 'in_progress' && $plan['voyage_status'] === 'planned') {
                $this->db->table('voyages')
                    ->where('id', $plan['voyage_id'])
                    ->set('lock_version', 'lock_version + 1', false)
                    ->update([
                        'status' => 'unloading',
                        'unloading_started_at' => $now,
                        'updated_by' => $actorUserId,
                        'updated_at' => $now,
                    ]);
            }

            $this->audit->record(
                'discharge_plan.transitioned',
                'discharge_plan',
                $planId,
                $actorUserId,
                (int) $plan['company_id'],
                (int) $plan['site_id'],
                ['status' => $plan['status']],
                ['status' => $target],
                $reason === '' ? null : $reason
            );

            $this->db->transCommit();

            return ['id' => $planId, 'status' => $target];
        } catch (\Throwable $exception) {
            $this->db->transRollback();
            throw $exception;
        }
    }

    public function startFilling(array $payload, int $actorUserId): array
    {
        $sourceTankIds = [];
        $sourceTankSeen = [];

        foreach (array_values($payload['source_tank_ids'] ?? []) as $sourceTankId) {
            if (
                ! is_int($sourceTankId)
                && ! (is_string($sourceTankId) && ctype_digit($sourceTankId))
            ) {
                throw new DomainException(
                    'La lista de tanques de procedencia contiene identificadores invÃ¡lidos.',
                    'INVALID_SOURCE_TANKS'
                );
            }

            $sourceTankId = (int) $sourceTankId;

            if ($sourceTankId <= 0 || isset($sourceTankSeen[$sourceTankId])) {
                throw new DomainException(
                    'Los tanques de procedencia deben ser positivos y no repetirse.',
                    'INVALID_SOURCE_TANKS'
                );
            }

            $sourceTankSeen[$sourceTankId] = true;
            $sourceTankIds[] = $sourceTankId;
        }

        $actualTankId = (int) $payload['actual_tank_id'];

        if ($sourceTankIds === [] || ! in_array($actualTankId, $sourceTankIds, true)) {
            throw new DomainException(
                'La procedencia debe incluir el tanque real seleccionado.',
                'SOURCE_TANKS_REQUIRED'
            );
        }

        $this->db->transException(true)->transBegin();

        try {
            $plan = $this->db->query(
                <<<'SQL'
SELECT dp.*, v.company_id, v.site_id
FROM discharge_plans dp
INNER JOIN voyages v ON v.id = dp.voyage_id
WHERE dp.id = ?
FOR UPDATE
SQL,
                [(int) $payload['plan_id']]
            )->getRowArray();

            if ($plan === null || $plan['status'] !== 'in_progress') {
                throw new DomainException(
                    'El plan no estÃ¡ en operaciÃ³n.',
                    'PLAN_NOT_IN_PROGRESS'
                );
            }

            // Las tinas son activos globales: no existe tubs.company_id.
            $tub = $this->db->query(
                'SELECT * FROM tubs WHERE id = ? FOR UPDATE',
                [(int) $payload['tub_id']]
            )->getRowArray();

            if (
                $tub === null
                || $tub['status'] !== 'available'
                || ! (bool) $tub['active']
            ) {
                throw new DomainException(
                    'La tina no estÃ¡ disponible.',
                    'TUB_NOT_AVAILABLE'
                );
            }

            $actualTank = $this->validateTankForPlan(
                (int) $plan['voyage_id'],
                $actualTankId
            );

            $plannedProduct = $this->db->table('discharge_plan_tanks')
                ->where([
                    'discharge_plan_id' => $plan['id'],
                    'voyage_tank_id' => $actualTankId,
                    'species_id' => (int) $payload['species_id'],
                ])
                ->groupStart()
                    ->where('size_grade_id', (int) $payload['size_grade_id'])
                    ->orWhere('size_grade_id', null)
                ->groupEnd()
                ->get()
                ->getRowArray();

            if ($plannedProduct === null) {
                throw new DomainException(
                    'La especie o talla no corresponde al tanque planeado.',
                    'PLAN_PRODUCT_MISMATCH'
                );
            }

            $size = $this->db->table('size_grades')
                ->where([
                    'id' => (int) $payload['size_grade_id'],
                    'active' => 1,
                ])
                ->get()
                ->getRowArray();

            if (
                $size === null
                || (
                    $size['species_id'] !== null
                    && (int) $size['species_id'] !== (int) $payload['species_id']
                )
            ) {
                throw new DomainException(
                    'La talla no corresponde a la especie capturada.',
                    'SIZE_SPECIES_MISMATCH'
                );
            }

            foreach ($sourceTankIds as $sourceTankId) {
                $this->validateTankForPlan((int) $plan['voyage_id'], $sourceTankId);
            }

            $sorterId = $this->validateOperator(
                isset($payload['sorter_user_id']) ? (int) $payload['sorter_user_id'] : null,
                'discharge.operate',
                (int) $plan['company_id'],
                (int) $plan['site_id'],
                'clasificador',
            );

            $labelerId = $this->validateOperator(
                isset($payload['labeler_user_id']) ? (int) $payload['labeler_user_id'] : null,

    private function referenceDate(?string $plannedStartAt): string
    {
        $today = date('Y-m-d');
        $planned = $plannedStartAt
            ? substr($plannedStartAt, 0, 10)
            : $today;

        return $planned < $today ? $today : $planned;
    }

    private function validateRsaForPlan(
        mixed $rsaId,
        array $voyage,
        ?string $plannedStartAt
    ): ?array {
        if ($rsaId === null || $rsaId === '' || (int) $rsaId <= 0) {

        }

        $rsa = $this->db->table('rsa_records')
            ->where([
                'id' => (int) $rsaId,
                'voyage_id' => (int) $voyage['id'],
                'status' => 'active',

            ->get()

        if ($rsa === null) {
            throw new DomainException(
                'El RSA no pertenece a la marea o no estÃ¡ activo.',
                'RSA_VOYAGE_MISMATCH'
            );
        }

        $referenceDate = $this->referenceDate($plannedStartAt);

        if ($rsa['issued_on'] !== null && $rsa['issued_on'] > $referenceDate) {
            throw new DomainException(
                'El RSA todavÃ­a no serÃ¡ vigente en la fecha de referencia del plan.',
                'PLAN_RSA_NOT_EFFECTIVE'

        }

        if ($rsa['expires_on'] !== null && $rsa['expires_on'] < $referenceDate) {
            throw new DomainException(
                'El RSA estarÃ¡ vencido en la fecha de referencia del plan.',
                'PLAN_RSA_EXPIRED'

    private function replaceTankLines(
        int $planId,
        array $voyage,
        array $tankLines,
        string $now
    ): void {
        $seen = [];
        $seenSequences = [];
        $plannedByTank = [];

        foreach ($tankLines as $index => $line) {
            $sequence = isset($line['sequence_number'])
                ? (int) $line['sequence_number']
                : $index + 1;

            if ($sequence <= 0 || isset($seenSequences[$sequence])) {
                throw new DomainException(
                    'Las secuencias del plan deben ser positivas y Ãºnicas.',
                    'DUPLICATE_PLAN_SEQUENCE'
                );
            }

            $seenSequences[$sequence] = true;

            $tank = $this->db->table('voyage_tanks')
                ->where([
                    'id' => (int) $line['voyage_tank_id'],
                    'voyage_id' => $voyage['id'],
                ])
                ->get()
                ->getRowArray();

            if ($tank === null) {
                throw new DomainException(
                    'Un tanque no pertenece a la marea.',
                    'PLAN_TANK_VOYAGE_MISMATCH'
                );
            }

            $composition = $this->validateComposition($line, $voyage);

            $fingerprint = implode(':', [
                (int) $line['voyage_tank_id'],
                (int) $line['species_id'],
                (int) ($line['size_grade_id'] ?? 0),
            ]);

            if (isset($seen[$fingerprint])) {
                throw new DomainException(
                    'La composiciÃ³n de tanque estÃ¡ duplicada.',
                    'DUPLICATE_TANK_COMPOSITION'
                );
            }

            $seen[$fingerprint] = true;
            $tankId = (int) $line['voyage_tank_id'];
            $plannedByTank[$tankId] = round(
                ($plannedByTank[$tankId] ?? 0)
                + (float) $line['planned_weight_kg'],
                3
            );

            if (
                $tank['capacity_kg'] !== null
                && $plannedByTank[$tankId] > (float) $tank['capacity_kg']
            ) {
                throw new DomainException(
                    'El tonelaje planeado supera la capacidad fotografiada del tanque.',
                    'PLAN_TANK_CAPACITY_EXCEEDED',
                    [
                        'tank_number' => $tank['tank_number'],
                        'capacity_kg' => (float) $tank['capacity_kg'],
                    ]
                );
            }

            $this->db->table('discharge_plan_tanks')->insert([
                'discharge_plan_id' => $planId,
                'voyage_tank_id' => $tankId,
                'species_id' => (int) $line['species_id'],
                'size_grade_id' => empty($line['size_grade_id'])
                    ? null
                    : (int) $line['size_grade_id'],
                'planned_weight_kg' => (float) $line['planned_weight_kg'],
                'msc_code' => $composition['msc_code'],
                'dolphin_safe_code' => $composition['dolphin_safe_code'],
                'freezing_temperature_c' => $composition['freezing_temperature_c'],
                'sequence_number' => $sequence,
                'status' => 'pending',
                'created_at' => $now,
                'updated_at' => $now,
            ]);
        }
    }

    private function loadTankLines(int $planId): array
    {
        return $this->db->table('discharge_plan_tanks dpt')
            ->select(
                'dpt.*, vt.tank_number, vt.capacity_kg, ' .
                's.code AS species_code, s.name AS species_name, ' .
                's.active AS species_active, sg.name AS size_name, ' .
                'sg.species_id AS size_species_id, sg.active AS size_active'

            ->join('voyage_tanks vt', 'vt.id = dpt.voyage_tank_id')
            ->join('species s', 's.id = dpt.species_id')
            ->join('size_grades sg', 'sg.id = dpt.size_grade_id', 'left')
            ->where('dpt.discharge_plan_id', $planId)
            ->orderBy('dpt.sequence_number')
            ->get()
            ->getResultArray();
    }

    private function buildReview(array $plan): array
    {
        $lines = isset($plan['tank_lines'])
            ? $plan['tank_lines']
            : $this->loadTankLines((int) $plan['id']);

        $referenceDate = $this->referenceDate($plan['planned_start_at'] ?? null);
        $rsa = null;

        if (! empty($plan['rsa_id'])) {
            $rsa = $this->db->table('rsa_records')
                ->where([
                    'id' => (int) $plan['rsa_id'],
                    'voyage_id' => (int) $plan['voyage_id'],
                    'status' => 'active',
                ])
                ->get()
                ->getRowArray();
        }

        $rsaOk = $rsa !== null
            && ($rsa['issued_on'] === null || $rsa['issued_on'] <= $referenceDate)
            && ($rsa['expires_on'] === null || $rsa['expires_on'] >= $referenceDate);

        $windowOk = ! (
            ! empty($plan['planned_start_at'])
            && ! empty($plan['planned_end_at'])
            && $plan['planned_end_at'] <= $plan['planned_start_at']
        );

        $hasLines = $lines !== [];
        $compositionOk = true;
        $duplicatesOk = true;
        $capacityOk = true;
        $seen = [];
        $plannedByTank = [];

        $mscValues = (array) $this->settings->get(
            'certification.msc_values',
            (int) $plan['company_id'],
            (int) $plan['site_id'],
            []
        );

        $dolphinValues = (array) $this->settings->get(
            'certification.dolphin_safe_values',
            (int) $plan['company_id'],
            (int) $plan['site_id'],
            []
        );

        foreach ($lines as $line) {
            $fingerprint = implode(':', [
                (int) $line['voyage_tank_id'],
                (int) $line['species_id'],
                (int) ($line['size_grade_id'] ?? 0),
            ]);

            if (isset($seen[$fingerprint])) {
                $duplicatesOk = false;
            }
            $seen[$fingerprint] = true;

            if (
                (float) $line['planned_weight_kg'] <= 0
                || ! is_numeric($line['freezing_temperature_c'])
                || ! in_array($line['msc_code'], $mscValues, true)
                || ! in_array($line['dolphin_safe_code'], $dolphinValues, true)
                || ! (bool) $line['species_active']
                || (
                    $line['size_grade_id'] !== null
                    && (
                        ! (bool) $line['size_active']
                        || (
                            $line['size_species_id'] !== null
                            && (int) $line['size_species_id'] !== (int) $line['species_id']
                        )
                    )
                )
            ) {
                $compositionOk = false;
            }

            $tankId = (int) $line['voyage_tank_id'];
            $plannedByTank[$tankId] =
                ($plannedByTank[$tankId] ?? 0)
                + (float) $line['planned_weight_kg'];

            if (
                $line['capacity_kg'] !== null
                && $plannedByTank[$tankId] > (float) $line['capacity_kg']
            ) {
                $capacityOk = false;
            }
        }

        $validations = [
            [
                'key' => 'rsa',
                'label' => 'RSA vigente para la fecha de referencia',
                'ok' => $rsaOk,
                'message' => $rsaOk
                    ? 'RSA vÃ¡lido.'
                    : 'Asocia un RSA activo y vigente para la fecha de referencia del plan.',
            ],
            [
                'key' => 'window',
                'label' => 'Ventana de planeaciÃ³n vÃ¡lida',
                'ok' => $windowOk,
                'message' => $windowOk
                    ? 'Fechas vÃ¡lidas.'
                    : 'La fecha de fin debe ser posterior a la fecha de inicio.',
            ],
            [
                'key' => 'lines',
                'label' => 'ComposiciÃ³n capturada',
                'ok' => $hasLines,
                'message' => $hasLines
                    ? sprintf('%d lÃ­nea(s) capturada(s).', count($lines))
                    : 'El plan no tiene composiciÃ³n.',
            ],
            [
                'key' => 'composition',
                'label' => 'Especies, tallas y certificaciones vÃ¡lidas',
                'ok' => $compositionOk,
                'message' => $compositionOk
                    ? 'ComposiciÃ³n vÃ¡lida.'
                    : 'Existe una lÃ­nea con especie, talla, MSC, Dolphin Safe, peso o temperatura invÃ¡lida.',
            ],
            [
                'key' => 'duplicates',
                'label' => 'Sin composiciones duplicadas',
                'ok' => $duplicatesOk,
                'message' => $duplicatesOk
                    ? 'No hay duplicados.'
                    : 'Existe una combinaciÃ³n tanque/especie/talla duplicada.',
            ],
            [
                'key' => 'capacity',
                'label' => 'Capacidad de tanques respetada',
                'ok' => $capacityOk,
                'message' => $capacityOk
                    ? 'Capacidades vÃ¡lidas.'
                    : 'El peso planeado supera la capacidad de al menos un tanque.',
            ],
        ];

        $canApprove = true;
        foreach ($validations as $validation) {
            if (! $validation['ok']) {
                $canApprove = false;
                break;
            }
        }

        return [
            'reference_date' => $referenceDate,
            'can_approve' => $canApprove,
            'validations' => $validations,
        ];
    }

    {

            if (! $validation['ok']) {
                return $validation;
            }
        }
        return [];
    }

    private function validateOperator(
        ?int $userId,
        string $requiredPermission,
        int $companyId,
        int $siteId,
        string $label
    ): ?int {
        if ($userId === null || $userId <= 0) {
            return null;
        }

        $operator = $this->db->table('users u')
            ->select('u.id')
            ->join('operational_user_scopes ous', 'ous.user_id = u.id')
            ->where([
                'u.id' => $userId,
                'u.active' => 1,
                'ous.company_id' => $companyId,
                'ous.active' => 1,
            ])
            ->groupStart()
                ->where('ous.site_id', $siteId)
                ->orWhere('ous.site_id', null)
            ->groupEnd()
            ->get()
            ->getRowArray();

        $user = $operator === null
            ? null
            : auth()->getProvider()->findById($userId);

        if ($user === null || ! $user->can($requiredPermission)) {
            throw new DomainException(
                'El usuario seleccionado no es un ' . $label . ' autorizado en este alcance.',
                'OPERATOR_SCOPE_MISMATCH'
            );
        }

        return $userId;
    }

    private function validateComposition(array $line, array $voyage): array
    {
        $mscCode = trim((string) ($line['msc_code'] ?? ''));
        $dolphinCode = trim((string) ($line['dolphin_safe_code'] ?? ''));
        $temperature = $line['freezing_temperature_c'] ?? null;

        $mscValues = (array) $this->settings->get(
            'certification.msc_values',
            (int) $voyage['company_id'],
            (int) $voyage['site_id'],
            []
        );

        $dolphinValues = (array) $this->settings->get(
            'certification.dolphin_safe_values',
            (int) $voyage['company_id'],
            (int) $voyage['site_id'],
            []
        );

        if (! in_array($mscCode, $mscValues, true)) {
            throw new DomainException(
                'ClasificaciÃ³n MSC invÃ¡lida para la composiciÃ³n.',
                'INVALID_MSC_CODE'
            );
        }

        if (! in_array($dolphinCode, $dolphinValues, true)) {
            throw new DomainException(
                'CategorÃ­a Dolphin Safe invÃ¡lida para la composiciÃ³n.',
                'INVALID_DOLPHIN_SAFE_CODE'
            );
        }

        if (! is_numeric($temperature)) {
            throw new DomainException(
                'La temperatura de congelaciÃ³n es obligatoria.',
                'FREEZING_TEMPERATURE_REQUIRED'
            );
        }

        if (
            ! isset($line['planned_weight_kg'])
            || ! is_numeric($line['planned_weight_kg'])
            || (float) $line['planned_weight_kg'] <= 0
        ) {
            throw new DomainException(
                'El tonelaje planeado debe ser positivo.',
                'INVALID_PLANNED_WEIGHT'
            );
        }

        $species = $this->db->table('species')
            ->where([
                'id' => (int) ($line['species_id'] ?? 0),
                'active' => 1,
            ])
            ->get()
            ->getRowArray();

        if ($species === null) {
            throw new DomainException(
                'La especie de la composiciÃ³n no existe.',
                'SPECIES_NOT_FOUND'
            );
        }

        if (! empty($line['size_grade_id'])) {
            $size = $this->db->table('size_grades')
                ->where([
                    'id' => (int) $line['size_grade_id'],
                    'active' => 1,
                ])
                ->get()
                ->getRowArray();

            if (
                $size === null
                || (
                    $size['species_id'] !== null
                    && (int) $size['species_id'] !== (int) $species['id']
                )
            ) {
                throw new DomainException(
                    'La talla no corresponde a la especie.',
                    'SIZE_SPECIES_MISMATCH'
                );
            }
        }

        return [
            'msc_code' => $mscCode,
            'dolphin_safe_code' => $dolphinCode,
            'freezing_temperature_c' => round((float) $temperature, 2),
        ];
    }

        if ($reading === null) {

        }
        if ($reading['weighing_id'] !== null) {
            throw new DomainException('La lectura de bÃ¡scula ya fue utilizada.', 'SCALE_READING_ALREADY_USED');
        }
        if ((int) $reading['scale_active'] !== 1 || (int) $reading['site_id'] !== (int) $filling['site_id'] || (int) $reading['scale_id'] !== $data->scaleId) {
            throw new DomainException('La lectura o la bÃ¡scula estÃ¡n fuera de la instalaciÃ³n operativa.', 'SCALE_SCOPE_MISMATCH');
        }
        if ((int) $reading['stable'] !== 1 || strtolower((string) $reading['unit']) !== 'kg' || ! $data->stable) {
            throw new DomainException('La captura no representa una lectura estable en kg.', 'UNSTABLE_WEIGHT');
        }

        $this->assertReadingActor($reading, $data->actorUserId);
        $this->assertReadingFreshness($reading, $filling);

        $normalizedTimestamp = $this->normalizeDeviceTimestamp($data->deviceTimestamp);
        $grossMatches = abs(round($data->grossWeightKg, 3) - (float) $reading['weight']) < 0.0005;
        $rawMatches = hash_equals((string) $reading['original_hash'], hash('sha256', $data->originalValue));
        $timestampMatches = $normalizedTimestamp === (string) $reading['device_timestamp'];
        $sourceMatches = $data->weightSource === $this->publicWeightSource((string) $reading['source']);

        if (! $grossMatches || ! $rawMatches || ! $timestampMatches || ! $sourceMatches) {
            throw new DomainException(
                'Los datos enviados no coinciden con la lectura capturada por el servidor.',
                'SCALE_READING_TAMPERED'
            );

    {

            if ($reading['actor_user_id'] === null || (int) $reading['actor_user_id'] !== $actorUserId) {
                throw new DomainException('La lectura fue capturada por otro usuario.', 'SCALE_READING_ACTOR_MISMATCH');
            }
            return;
        }

        $client = service('apiClientContext')->client();
        $currentReference = $client === null ? null : (string) $client['client_id'];
        $readingReference = $reading['actor_reference'] === null ? null : (string) $reading['actor_reference'];
        if ($currentReference === null || $readingReference === null || ! hash_equals($readingReference, $currentReference)) {
            throw new DomainException('La lectura fue capturada por otro cliente de integraciÃ³n.', 'SCALE_READING_ACTOR_MISMATCH');
        }
    }

    /** @param array<string, mixed> $reading @param array<string, mixed> $filling */
    private function assertReadingFreshness(array $reading, array $filling): void
    {
        $timezone = new \DateTimeZone(config('App')->appTimezone);
        try {
            $receivedAt = new \DateTimeImmutable((string) $reading['received_at'], $timezone);
        } catch (\Throwable $exception) {
            throw new DomainException('La hora de recepciÃ³n de la lectura no es vÃ¡lida.', 'INVALID_SCALE_TIMESTAMP', [], 0, $exception);
        }

        $maxAge = (int) $this->settings->get(
            'scale.reading_max_age_seconds',
            (int) $filling['company_id'],
            (int) $filling['site_id'],
            120,
        );
        $maxAge = max(5, min(3600, $maxAge));
        $ageSeconds = (new \DateTimeImmutable('now', $timezone))->getTimestamp() - $receivedAt->getTimestamp();
        if ($ageSeconds > $maxAge || $ageSeconds < -30) {
            throw new DomainException(
                'La lectura de bÃ¡scula expirÃ³; solicita una nueva captura estable.',
                'STALE_SCALE_READING',
                ['max_age_seconds' => $maxAge]
            );
        }
    }

    private function publicWeightSource(string $source): string
    {
        if ($source === 'simulator') {
            return 'simulator';
        }
        if ($source === 'middleware' || str_starts_with($source, 'local_middleware:')) {
            return 'middleware';
        }

        return 'scale';
    }

    private function normalizeDeviceTimestamp(?string $value): ?string
    {
        if ($value === null || trim($value) === '') {
            return null;
        }
        try {
            return (new \DateTimeImmutable($value))
                ->setTimezone(new \DateTimeZone(config('App')->appTimezone))
                ->format('Y-m-d H:i:s');
        } catch (\Throwable $exception) {
            throw new DomainException('La fecha de la bÃ¡scula no es vÃ¡lida.', 'INVALID_SCALE_TIMESTAMP', [], 0, $exception);
        }
    }

    private function assertPrinter(?int $printerId, int $siteId): ?array
    {
        $driver = strtolower((string) env('integrations.printer.driver', 'pdf'));
        if ($printerId === null) {
            if (! in_array($driver, ['pdf', 'browser'], true)) {
                throw new DomainException('La impresiÃ³n fÃ­sica requiere seleccionar una impresora.', 'PRINTER_ID_REQUIRED');
            }

            return null;
        }

        $printer = $this->db->table('printers')->where(['id' => $printerId, 'site_id' => $siteId, 'active' => 1])->get()->getRowArray();
        if ($printer === null) {
            throw new DomainException('La impresora no estÃ¡ activa en esta instalaciÃ³n.', 'PRINTER_NOT_AVAILABLE');
        }

        $compatibleAdapters = match ($driver) {
            'pdf' => ['pdf'],
            'browser' => ['browser'],
            'middleware' => ['middleware', 'zpl'],
            'thermal', 'zpl' => ['zpl', 'middleware'],
            default => [],
        };
        if (! in_array((string) $printer['adapter_key'], $compatibleAdapters, true)) {
            throw new DomainException('La impresora seleccionada no corresponde al driver activo de esta estaciÃ³n.', 'PRINTER_DRIVER_MISMATCH');
        }

        return $printer;
    }

    private function assertTemplateCompatible(array $template, ?array $printer): void
    {
        $adapter = (string) ($printer['adapter_key'] ?? strtolower((string) env('integrations.printer.driver', 'pdf')));

            throw new DomainException('La impresiÃ³n PDF o de navegador requiere una plantilla HTML.', 'LABEL_TEMPLATE_PRINTER_MISMATCH');

        if ($adapter === 'zpl' && $format !== 'zpl') {

        }
    }

    public function restrict(int $locationId, array $payload, int $actorUserId): array
    {
        $location = $this->findStack($locationId);
        service('scopeContext')->assertWarehouse((int) $location['warehouse_id']);

        $type = trim((string) ($payload['restriction_type'] ?? ''));
        $reason = trim((string) ($payload['reason'] ?? ''));
        $notes = trim((string) ($payload['notes'] ?? ''));

        $allowedTypes = ['humidifier', 'maintenance', 'obstacle', 'reserved', 'other'];

        if (! in_array($type, $allowedTypes, true)) {
            throw new DomainException(
                'Selecciona un tipo de restricciÃ³n vÃ¡lido.',
                'INVALID_LOCATION_RESTRICTION_TYPE'
            );
        }

        if ($reason === '') {
            throw new DomainException(
                'Indica el motivo de la inhabilitaciÃ³n.',
                'LOCATION_RESTRICTION_REASON_REQUIRED'
            );
        }

        if (mb_strlen($reason) > 150) {
            throw new DomainException(
                'El motivo no puede superar 150 caracteres.',
                'LOCATION_RESTRICTION_REASON_TOO_LONG'
            );
        }

        $activeStock = $this->activeStockCount($locationId);

        if ($activeStock > 0) {
            throw new DomainException(
                'No se puede inhabilitar una estiba que actualmente contiene inventario.',
                'LOCATION_RESTRICTION_HAS_STOCK'
            );
        }

        $alreadyBlocked = $this->db->table('location_restrictions')
            ->where('location_id', $locationId)
            ->where('active', 1)
            ->countAllResults();

        if ($alreadyBlocked > 0) {
            throw new DomainException(
                'La ubicaciÃ³n ya tiene una restricciÃ³n activa.',
                'LOCATION_RESTRICTION_ALREADY_ACTIVE'
            );
        }

        $this->db->table('location_restrictions')->insert([
            'location_id' => $locationId,
            'restriction_type' => $type,
            'reason' => $reason,
            'notes' => $notes === '' ? null : $notes,
            'active' => 1,
            'created_by' => $actorUserId > 0 ? $actorUserId : null,
            'created_at' => date('Y-m-d H:i:s'),
        ]);

        return [
            'id' => (int) $this->db->insertID(),
            'location_id' => $locationId,
            'status' => 'blocked',
        ];
    }

        service('scopeContext')->assertWarehouse((int) $location['warehouse_id']);

        $active = $this->db->table('location_restrictions')
            ->where('location_id', $locationId)
            ->where('active', 1)
            ->get()
            ->getResultArray();

        if ($active === []) {
            throw new DomainException(
                'La ubicaciÃ³n no tiene una restricciÃ³n activa.',
                'LOCATION_RESTRICTION_NOT_FOUND'
            );
        }

        $this->db->table('location_restrictions')
            ->where('location_id', $locationId)
            ->where('active', 1)
            ->update([
                'active' => 0,
                'released_by' => $actorUserId > 0 ? $actorUserId : null,
                'released_at' => date('Y-m-d H:i:s'),
            ]);

        return [
            'location_id' => $locationId,
            'status' => 'released',
        ];
    }

    private function findChamber(int $chamberId): array
    {
        $row = $this->db->table('chambers c')
            ->select('c.id, c.warehouse_id, c.code, c.name, w.code AS warehouse_code, w.name AS warehouse_name')
            ->join('warehouses w', 'w.id = c.warehouse_id')
            ->where('c.id', $chamberId)
            ->where('c.active', 1)
            ->get()
            ->getRowArray();

        if ($row === null) {
            throw new DomainException(
                'CÃ¡mara no encontrada o inactiva.',
                'CHAMBER_NOT_FOUND'
            );
        }

        return $row;
    }

    private function findStack(int $locationId): array
    {
        $row = $this->db->table('locations')
            ->select('id, warehouse_id, chamber_id, code, name, location_type, active')
            ->where('id', $locationId)
            ->get()
            ->getRowArray();

        if (
            $row === null
            || (int) $row['active'] !== 1
            || ! in_array((string) $row['location_type'], ['stack_level', 'stack'], true)
        ) {
            throw new DomainException(
                'La ubicaciÃ³n seleccionada no es una estiba activa.',
                'STACK_LOCATION_NOT_FOUND'
            );
        }

        return $row;
    }

    private function activeStockCount(int $locationId): int
    {
        return $this->db->table('stock_units')
            ->where('location_id', $locationId)
            ->where('on_hand_weight_kg >', 0)
            ->whereNotIn('status', ['in_transit', 'dispatched', 'exhausted'])
            ->countAllResults();
    }

    private function extractSectorNumber(string $code): int
    {
        if (preg_match('/(?:^|-)(?:F|P)0*(\d+)(?:-|$)/i', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        if (preg_match('/(\d+)(?!.*\d)/', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        return 0;
    }

    private function extractPositionNumber(string $code): int
    {
        if (preg_match('/(?:POS|POSICION)[-_]?0*(\d+)/i', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        if (preg_match('/-P0*(\d+)(?:-|$)/i', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        if (preg_match('/(\d+)(?!.*\d)/', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        return 0;
    }

    private function extractStackNumber(string $code): int
    {
        if (preg_match('/(?:^|-)E0*(\d+)(?:-|$)/i', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        if (preg_match('/(\d+)(?!.*\d)/', $code, $matches) === 1) {
            return (int) $matches[1];
        }

        return 0;
    }

    const options = ['<option value="">Seleccionaâ€¦</option>', ...rows.map((tank) => `<option value="${Number(tank.id)}">${String(tank.tank_number).replace(/[&<>"']/g, '')}</option>`)].join('');
    field.actualTank.innerHTML = options;
    field.sources.innerHTML = rows.map((tank) => `<option value="${Number(tank.id)}">${String(tank.tank_number).replace(/[&<>"']/g, '')}</option>`).join('');
    if (globalThis.jQuery) { jQuery(field.actualTank).trigger('change.select2'); jQuery(field.sources).trigger('change.select2'); }
    field.rsa.value = selectedPlan()?.rsa_code || 'Sin RSA asociado';
  }

  function appliedTare() {
    if (field.tareSource.value === 'estimated') return Number(data.estimatedTare || 0);
    if (field.tareSource.value === 'painted') return Number(field.declaredTare.value || 0);
    return Number(selectedTub()?.current_tare_kg || 0);
  }

  function calculate() {
    const gross = Number(field.gross.value || 0);
    const tare = appliedTare();
    const physical = Math.max(0, gross - tare);
    const penalty = physical * Math.max(0, Math.min(100, Number(field.penalty.value || 0))) / 100;
    byId('grossDisplay').textContent = gross.toFixed(3);
    byId('tareResult').textContent = `${tare.toFixed(3)} kg`;
    byId('physicalNetResult').textContent = `${physical.toFixed(3)} kg`;
    byId('penaltyResult').textContent = `${penalty.toFixed(3)} kg`;
    byId('acceptedResult').textContent = `${Math.max(0, physical - penalty).toFixed(3)} kg`;
  }

  async function checkScale() {
    const status = byId('scaleStatus');
    try {
      const result = await GMar.api('/internal/v1/scale/status');
      const connected = Boolean(result.status?.connected);
      status.className = `status-pill ${connected ? 'online' : 'warning'}`;
      status.querySelector('span:last-child').textContent = `BÃ¡scula: ${result.status?.status || 'sin estado'}`;
    } catch (error) {
      status.className = 'status-pill offline';
      status.querySelector('span:last-child').textContent = 'BÃ¡scula: no disponible';
    }
  }

  field.plan.addEventListener('change', populateTanks);
  field.actualTank.addEventListener('change', () => {
    const id = field.actualTank.value;
    const selected = new Set(Array.from(field.sources.selectedOptions).map((option) => option.value));
    if (id) selected.add(id);
    if (globalThis.jQuery) jQuery(field.sources).val(Array.from(selected)).trigger('change');
  });
  field.pending.addEventListener('change', () => {
    const pending = data.pending.find((item) => Number(item.id) === Number(field.pending.value));
    fillingId = pending ? Number(pending.id) : null;
    if (!pending) return;
    selectValue(field.plan, pending.discharge_plan_id);
    populateTanks();
    selectValue(field.tub, pending.tub_id);
    selectValue(field.actualTank, pending.actual_voyage_tank_id);
    selectValue(field.species, pending.species_id);
    selectValue(field.size, pending.size_grade_id);
    if (globalThis.jQuery) jQuery(field.sources).val((data.sourcesByFilling || {})[pending.id] || [pending.actual_voyage_tank_id]).trigger('change');
    calculate();
    byId('lotStatus').textContent = 'Se resolverÃ¡ con los componentes del llenado';
  });
  field.tub.addEventListener('change', calculate);
  field.tareSource.addEventListener('change', () => {
    field.declaredTare.disabled = field.tareSource.value !== 'painted';
    if (field.tareSource.value === 'estimated') field.reason.required = true;
    calculate();
  });
  field.declaredTare.addEventListener('input', calculate);
  field.penalty.addEventListener('input', calculate);
  field.gross.addEventListener('input', calculate);
  field.weightSource.addEventListener('change', () => {
    const manual = field.weightSource.value === 'manual';
    field.gross.readOnly = !manual;
    field.reason.required = manual || field.tareSource.value !== 'catalog';
    if (manual) { reading = null; byId('readingStatus').textContent = 'Captura manual pendiente de motivo'; }
  });

  byId('prepareFilling')?.addEventListener('click', async () => {
    const payload = {
      plan_id: Number(field.plan.value), tub_id: Number(field.tub.value), actual_tank_id: Number(field.actualTank.value),
      source_tank_ids: Array.from(field.sources.selectedOptions).map((option) => Number(option.value)),
      species_id: Number(field.species.value), size_grade_id: Number(field.size.value),
      sorter_user_id: field.sorter.value ? Number(field.sorter.value) : null,
      labeler_user_id: field.labeler.value ? Number(field.labeler.value) : null,
      observations: field.observations.value || null
    };
    if (!payload.plan_id || !payload.tub_id || !payload.actual_tank_id || !payload.species_id || !payload.size_grade_id || payload.source_tank_ids.length === 0) {
      return Swal.fire({ icon: 'warning', text: 'Completa plan, tina, tanque real, procedencias, especie y talla.' });
    }
    const button = byId('prepareFilling'); button.disabled = true;
    try {
      const started = await GMar.api('/internal/v1/fillings', { method: 'POST', body: JSON.stringify(payload) });
      const completed = await GMar.api(`/internal/v1/fillings/${started.id}/complete`, { method: 'POST', body: '{}' });
      fillingId = Number(completed.id);
      await Swal.fire({ icon: 'success', text: `Tina preparada. Llenado #${fillingId}`, timer: 1500, showConfirmButton: false });
    } catch (error) {
      await Swal.fire({ icon: 'error', title: error.code || 'No se pudo preparar', text: error.message });
    } finally { button.disabled = !navigator.onLine; }
  });

  byId('readScale').addEventListener('click', async () => {
    const button = byId('readScale'); button.disabled = true;
    try {
      reading = await GMar.api('/internal/v1/scale/read-stable', { method: 'POST', body: JSON.stringify({ scale_id: data.scale?.id ? Number(data.scale.id) : null }) });
      field.gross.value = Number(reading.weight).toFixed(3);
      field.weightSource.value = reading.weight_source || (String(reading.source).startsWith('simulator') ? 'simulator' : 'middleware');
      byId('readingStatus').textContent = `${reading.stable ? 'Estable' : 'Inestable'} Â· ${reading.scale_identifier} Â· ${reading.captured_at}`;
      calculate();
    } catch (error) {
      await Swal.fire({ icon: 'error', title: 'BÃ¡scula no disponible', text: error.message });
    } finally { button.disabled = !navigator.onLine; }
  });

  byId('confirmWeighing').addEventListener('click', async () => {
    const manual = field.weightSource.value === 'manual';
    if (!fillingId || Number(field.gross.value) <= 0) return Swal.fire({ icon: 'warning', text: 'Prepara o selecciona un llenado y captura el peso.' });
    if (!manual && !reading?.stable) return Swal.fire({ icon: 'warning', text: 'Se requiere una lectura estable de bÃ¡scula.' });
    if ((manual || field.tareSource.value !== 'catalog' || Number(field.penalty.value) > 0) && !field.reason.value.trim()) return Swal.fire({ icon: 'warning', text: 'Indica el motivo de contingencia o castigo.' });
    const plan = selectedPlan();
    const payload = {
      filling_id: fillingId, rsa_id: plan?.rsa_id ? Number(plan.rsa_id) : null,
      scale_id: data.scale?.id ? Number(data.scale.id) : null,
      scale_reading_id: manual ? null : Number(reading.id), gross_weight_kg: Number(field.gross.value),
      declared_tare_kg: field.tareSource.value === 'painted' ? Number(field.declaredTare.value) : null,
      tare_source: field.tareSource.value, penalty_percent: Number(field.penalty.value || 0), weight_source: field.weightSource.value,
      stable: manual ? false : Boolean(reading?.stable), original_value: reading?.original_value || null,
      device_timestamp: reading?.captured_at || null, reason_code: field.reason.value.trim() || null, observations: field.observations.value.trim() || null
    };
    const button = byId('confirmWeighing'); button.disabled = true;
    try {
      const weighing = await GMar.api('/internal/v1/weighings', { method: 'POST', body: JSON.stringify(payload) });
      let printText = 'Etiqueta no solicitada.';
      if (byId('printAfter').checked && data.template?.id) {
        const printed = await GMar.api('/internal/v1/labels/print', { method: 'POST', body: JSON.stringify({ weighing_id: Number(weighing.id), template_id: Number(data.template.id), printer_id: data.printer?.id ? Number(data.printer.id) : null }) });
        printText = `Etiqueta ${printed.label.code}: ${printed.print.status}`;
        if (printed.print?.document_html) GMar.printHtml(printed.print.document_html, printed.label.code);
        if (printed.print?.download_available) printText += `<br><a class="btn btn-sm btn-outline-primary mt-2" href="/operacion/etiquetas/trabajos/${Number(printed.print_job_id)}/archivo">Descargar PDF</a>`;
      }
      await Swal.fire({ icon: 'success', title: `Pesaje ${weighing.folio}`, html: `<strong>${Number(weighing.accepted_net_kg).toFixed(3)} kg aceptados</strong><br>${printText}`, confirmButtonColor: '#0d5f82' });
      window.location.reload();
    } catch (error) {
      await Swal.fire({ icon: 'error', title: error.code || 'OperaciÃ³n rechazada', text: error.message });
    } finally { button.disabled = !navigator.onLine; }
  });

  calculate();
  checkScale();
})();
